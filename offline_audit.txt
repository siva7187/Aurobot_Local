     1	# vision_worker.py
     2	"""
     3	Runs face + emotion detection in a SEPARATE OS PROCESS.
     4	
     5	Why a separate process: DeepFace/MTCNN inference is CPU-heavy. Running it as
     6	a thread in the same process as the LiveKit voice session means it competes
     7	for the same cores (and the same GIL for pure-Python bookkeeping) as your
     8	real-time audio pipeline — a common source of audio stutter/lag. A separate
     9	process gets its own memory space and OS scheduling, so a slow detection
    10	pass can't directly stall audio.
    11	
    12	PLUGGABLE FRAME SOURCE (hardware-deployment groundwork):
    13	  source="webcam" (default) — cv2.VideoCapture(0), for laptop/dev use.
    14	  source="remote"           — reads frames from `frame_queue` instead, which
    15	                               the caller feeds from wherever frames actually
    16	                               come from (e.g. a LiveKit video track
    17	                               subscribed to a remote embedded device's
    18	                               camera). This worker never needs to know
    19	                               whether the camera is local or on a device
    20	                               across the room — that decision lives entirely
    21	                               in agent.py's setup code, not here.
    22	
    23	Output communication is one-way, worker -> main, over a multiprocessing.Queue
    24	of plain dicts (picklable), same as before.
    25	"""
    26	
    27	import os
    28	import queue
    29	import time
    30	
    31	
    32	def _process_frame(frame, frame_width, detector_backend, cv2_module):
    33	    h, w = frame.shape[:2]
    34	    if w > frame_width:
    35	        scale = frame_width / w
    36	        frame = cv2_module.resize(frame, (frame_width, int(h * scale)))
    37	    return cv2_module.cvtColor(frame, cv2_module.COLOR_BGR2RGB)

    39	
    40	def run(
    41	    result_queue,
    42	    poll_seconds: float = 2.5,
    43	    frame_width: int = 320,
    44	    source: str = "webcam",
    45	    frame_queue=None,          # required when source == "remote"
    46	    detector_backend: str = "mtcnn",
    47	) -> None:
    48	    os.environ.setdefault("TF_CPP_MIN_LOG_LEVEL", "3")
    49	    os.environ.setdefault("TF_ENABLE_ONEDNN_OPTS", "0")
    50	
    51	    # Opt-in: pin THIS process to CPU so the GPU's VRAM stays free (offline mode
    52	    # gives it to the local LLM). Only affects this child process, and must be set
    53	    # before TensorFlow is imported. Default ("auto") leaves behaviour unchanged.
    54	    if os.environ.get("AUROBOT_VISION_DEVICE", "auto").lower() == "cpu":
    55	        os.environ["CUDA_VISIBLE_DEVICES"] = "-1"
    56	
    57	    # GPU setup MUST happen before any TensorFlow op runs (including DeepFace's
    58	    # import-time model construction) — memory growth can't be changed once a
    59	    # GPU has been initialized. Without this, TensorFlow grabs ALL VRAM upfront
    60	    # by default, which can itself cause a noticeable stall on first inference
    61	    # and can starve anything else on the GPU.
    62	    import tensorflow as tf
    63	    gpus = tf.config.list_physical_devices("GPU")
    64	    if gpus:
    65	        try:
    66	            for gpu in gpus:
    67	                tf.config.experimental.set_memory_growth(gpu, True)
    68	            print(f"[vision_worker] GPU acceleration enabled: {[g.name for g in gpus]}")
    69	        except RuntimeError as e:
    70	            print(f"[vision_worker] GPU found but memory-growth config failed: {e}")
    71	    else:
    72	        print("[vision_worker] No GPU visible to TensorFlow — running vision on CPU")
    73	
    74	    import cv2
    75	    from deepface import DeepFace
    76	
    77	    cap = None
    78	    if source == "webcam":
    79	        cap = cv2.VideoCapture(0)
    80	        if not cap.isOpened():
    81	            result_queue.put({"error": "camera_not_accessible"})
    82	            return
    83	    elif source == "remote":
    84	        if frame_queue is None:
    85	            result_queue.put({"error": "remote source selected but no frame_queue provided"})
    86	            return
    87	    else:
    88	        result_queue.put({"error": f"unknown vision source: {source}"})
    89	        return
    90	
    91	    last_error = None
    92	
    93	    while True:
    94	        if source == "webcam":
    95	            ret, frame = cap.read()
    96	            if not ret:
    97	                continue
    98	        else:  # remote
    99	            try:
   100	                frame = frame_queue.get(timeout=poll_seconds * 2)
   101	            except queue.Empty:
   102	                continue  # no frame arrived from the remote device recently — just wait again
   103	
   104	        try:
   105	            rgb_frame = _process_frame(frame, frame_width, detector_backend, cv2)
   106	
   107	            t0 = time.time()
   108	            result = DeepFace.analyze(
   109	                rgb_frame,
   110	                actions=["emotion"],
   111	                enforce_detection=False,
   112	                detector_backend=detector_backend,
   113	            )
   114	            analyze_ms = (time.time() - t0) * 1000
   115	
   116	            emotion_scores = result[0]["emotion"]
   117	            result_queue.put({"emotion_scores": emotion_scores, "analyze_ms": analyze_ms})
   118	            last_error = None
   119	
   120	        except Exception as e:
   121	            if str(e) != last_error:
   122	                result_queue.put({"error": str(e)})
   123	                last_error = str(e)
   124	
   125	        if source == "webcam":
   126	            time.sleep(poll_seconds)
   127	        # for "remote", pacing is naturally set by how often frames arrive in frame_queue
